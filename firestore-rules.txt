rules_version = '2';

// ==========================================================
// CORTEX UNDER CONSTRUCTION — FIRESTORE SECURITY RULES
// ----------------------------------------------------------
// These rules decide who can read and write comments,
// reactions and display names. Copy ALL of this text into
// Firebase → Firestore Database → Rules, replacing what is
// there, then click "Publish".
//
// ONE THING TO CHANGE: paste your author ID (shown on your
// login page after you sign in) between the quotes below.
// ==========================================================

service cloud.firestore {
  match /databases/{database}/documents {

    // ▼▼▼ PASTE YOUR AUTHOR ID BETWEEN THE QUOTES ▼▼▼
    function authorUid() {
      return "PASTE-YOUR-AUTHOR-ID-HERE";
    }
    // ▲▲▲ ─────────────────────────────────────── ▲▲▲

    function signedIn() {
      return request.auth != null;
    }

    function isAuthor() {
      return signedIn() && request.auth.uid == authorUid();
    }

    // Display names: 2–30 characters. Readers can't use names
    // containing cortex, admin, author or moderator.
    function validName(name) {
      return name is string
        && name.size() >= 2
        && name.size() <= 30
        && (isAuthor() || !name.lower().matches('.*(cortex|admin|author|moderator).*'));
    }

    // ---------- Display names (one per account) ----------
    match /profiles/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;

      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name', 'lastCommentAt'])
        && request.resource.data.keys().hasAll(['name', 'lastCommentAt'])
        && validName(request.resource.data.name)
        && request.resource.data.lastCommentAt is timestamp
        && request.resource.data.lastCommentAt <= request.time;

      allow update: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name', 'lastCommentAt'])
        && validName(request.resource.data.name)
        && (request.resource.data.lastCommentAt == resource.data.lastCommentAt
            || request.resource.data.lastCommentAt == request.time);

      allow delete: if false;
    }

    match /articles/{articleId} {

      // ---------- Reactions: one per person per article ----------
      match /reactions/{uid} {
        allow read: if true;

        allow create, update: if signedIn() && request.auth.uid == uid
          && articleId.matches('[a-z0-9-]{1,60}')
          && request.resource.data.keys().hasOnly(['type', 'updatedAt'])
          && request.resource.data.keys().hasAll(['type', 'updatedAt'])
          && request.resource.data.type in ['heart', 'curious', 'learned', 'skeptical', 'dislike']
          && request.resource.data.updatedAt == request.time;

        allow delete: if signedIn() && request.auth.uid == uid;
      }

      // ---------- Comments ----------
      match /comments/{commentId} {
        allow read: if true;

        // Anyone with an account can comment, at most once every 30 seconds.
        // The author has no waiting time and gets the "Author" badge.
        allow create: if signedIn()
          && articleId.matches('[a-z0-9-]{1,60}')
          && request.resource.data.keys().hasOnly(['uid', 'name', 'text', 'createdAt', 'isAuthor'])
          && request.resource.data.keys().hasAll(['uid', 'name', 'text', 'createdAt', 'isAuthor'])
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.text is string
          && request.resource.data.text.size() >= 2
          && request.resource.data.text.size() <= 2000
          && request.resource.data.createdAt == request.time
          && request.resource.data.isAuthor == isAuthor()
          && request.resource.data.name == getAfter(/databases/$(database)/documents/profiles/$(request.auth.uid)).data.name
          && (isAuthor()
              || (get(/databases/$(database)/documents/profiles/$(request.auth.uid)).data.lastCommentAt
                    < request.time - duration.value(30, 's')
                  && getAfter(/databases/$(database)/documents/profiles/$(request.auth.uid)).data.lastCommentAt
                    == request.time));

        // Comments can't be edited after posting.
        allow update: if false;

        // Readers can delete their own comments; the author can delete any comment.
        allow delete: if isAuthor()
          || (signedIn() && resource.data.uid == request.auth.uid);
      }
    }
  }
}
